        -- TF-IDF
            Let's also quickly see the code in action. We are going to use
            sklearn for this. To visualize the results, we are going to
            use Pandas data frames.

            import pandas as pd
            from sklearn.feature_extraction.text import TfidfVectorizer

            corpus = [
            'cats and dogs are pets.',
            'cats and dogs are pet animals though I prefer dogs.
            Dogs obey our commands, can be trained easily and play with
            us all the time.',
            'And this is the third one.',
            'Horses are also pets',
            ]

            vectorizer = TfidfVectorizer()
            X = vectorizer.fit_transform(corpus)
            vectorizer.get_feature_names_out()
            feature_names = vectorizer.get_feature_names_out()

            tfidf_df = pd.DataFrame(data=X.toarray(),columns=feature_names)
            tfidf_df = pd.DataFrame(data=X.toarray(), columns=feature_names)

            print(tfidf_df)
